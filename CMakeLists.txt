cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(NesDQ VERSION 1.0.5 LANGUAGES C CXX)
include(FetchContent)
if(ANDROID)
 set(SDL_SHARED ON CACHE BOOL "" FORCE)
 set(SDL_STATIC OFF CACHE BOOL "" FORCE)
 add_compile_options("$<$<CONFIG:Debug>:-O2>")
else()
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.2.28/SDL3-3.2.28.zip URL_HASH SHA256=24a30069af514a6c6b773bdc8ccca8b321661b251381acc1daeebf8c8f4a109a)
FetchContent_MakeAvailable(SDL3)
add_library(binjnes STATIC third_party/binjnes/src/emulator.c third_party/binjnes/src/cartdb.c third_party/binjnes/src/common.c)
set_target_properties(binjnes PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_include_directories(binjnes PUBLIC third_party/binjnes/src)
if(MSVC)
 target_compile_definitions(binjnes PRIVATE BINJNES_MSVC _CRT_SECURE_NO_WARNINGS)
 target_compile_options(binjnes PRIVATE /utf-8 /wd4146 /wd4098)
else()
 target_compile_definitions(binjnes PRIVATE BINJNES_CLANG)
endif()
if(ANDROID)
 add_library(nesdq SHARED app/main.cpp app/core_bridge.c)
 set_target_properties(nesdq PROPERTIES OUTPUT_NAME main)
 set_target_properties(binjnes PROPERTIES POSITION_INDEPENDENT_CODE ON)
else()
 add_executable(nesdq app/main.cpp app/core_bridge.c)
 if(WIN32)
  enable_language(RC)
  target_sources(nesdq PRIVATE app/nesdq.rc)
 endif()
endif()
set_property(TARGET nesdq PROPERTY C_STANDARD 11)
target_compile_features(nesdq PRIVATE cxx_std_20)
target_include_directories(nesdq PRIVATE third_party/picosha2)
target_link_libraries(nesdq PRIVATE SDL3::SDL3 binjnes)
if(WIN32)
 target_link_libraries(nesdq PRIVATE gdi32)
endif()
if(MSVC)
 target_compile_definitions(nesdq PRIVATE BINJNES_MSVC)
 target_compile_options(nesdq PRIVATE /W4 /utf-8)
else()
 target_compile_definitions(nesdq PRIVATE BINJNES_CLANG)
endif()
install(TARGETS nesdq RUNTIME DESTINATION .)
if(WIN32)
 add_executable(nesdq_picker_check EXCLUDE_FROM_ALL tests/check_rom_picker.cpp app/core_bridge.c)
 target_compile_features(nesdq_picker_check PRIVATE cxx_std_20)
 set_property(TARGET nesdq_picker_check PROPERTY C_STANDARD 11)
 target_include_directories(nesdq_picker_check PRIVATE third_party/picosha2)

 target_compile_definitions(nesdq_picker_check PRIVATE BINJNES_MSVC)
 target_compile_options(nesdq_picker_check PRIVATE /utf-8)
endif()
install(DIRECTORY DESTINATION Rom)
install(FILES third_party/binjnes/LICENSE third_party/binjnes/PALETTE-LICENSE.txt DESTINATION licenses/binjnes)
install(FILES third_party/picosha2/LICENSE DESTINATION licenses/picosha2)
install(FILES "${sdl3_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses/SDL3)

install(DIRECTORY third_party/sdl3/ DESTINATION licenses/SDL3-source)
install(FILES cmake/PatchSDLSwitch.cmake DESTINATION licenses/SDL3-source)
